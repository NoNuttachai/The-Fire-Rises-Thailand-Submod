SIA_for_loop_reduce_specialist_in_state = {
	for_each_loop = { 
		array = SIA_map_var 
		value = v 
		
		var:v = {
			if = {
				limit = { #if there's a specialist and the change var is not 0, find and subtract
					check_variable = { SIA_specialists_in_state > 0 }
					NOT = {
						check_variable = { change_temp = 0 }
					}
				}
				subtract_from_variable = { SIA_specialists_in_state = 1 }
				add_to_temp_variable = { SIA.change_temp = 1 }
				add_to_variable = { SIA.SIA_specialists_var_temp = SIA_specialists_in_state }
			}
		}
	}
	if = {
		limit = { #if somehow the change var is alive and there's still a specialist somewhere, continue, if not then go to next effect
			NOT = {
				check_variable = { change_temp = 0 }
			}
			check_variable = { SIA_specialists_var_temp > 0 }
		}
		set_temp_variable = { SIA_specialists_var_temp = 0 }
		SIA_for_loop_reduce_specialist_in_state = yes
	}
	else = {
		#stop, just continue ahead
	}
}
