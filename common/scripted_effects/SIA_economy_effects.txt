	set_temp_variable = { current_year_net_gain = SIA_environment_current_progress }
	subtract_from_temp_variable = { current_year_net_gain = SIA_environment_previous_year_progress }
	
	#giant list to see how well you did
	if = { #excellent
		limit = {
			check_variable = { current_year_net_gain > 12 }
		}
		set_variable = { SIA_environment_prev_year_status = 5 }
	}
	else_if = { #good
		limit = {
			check_variable = { current_year_net_gain > 8 }
		}
		set_variable = { SIA_environment_prev_year_status = 4 }
	}
	else_if = { #okay
		limit = {
			check_variable = { current_year_net_gain > 4 }
		}
		set_variable = { SIA_environment_prev_year_status = 3 }
	}
	else_if = { #bad
		limit = {
			check_variable = { current_year_net_gain > 2 }
		}
		set_variable = { SIA_environment_prev_year_status = 2 }
	}
	else = { #Ministry of Environment goes sad
		
		set_variable = { SIA_environment_prev_year_status = 1 }
	}
	#use this for standard in next year tick
	set_variable = { SIA_environment_previous_previous_year_progress = SIA_environment_previous_year_progress }
	set_variable = { SIA_environment_previous_year_progress = SIA_environment_current_progress }
	set_variable = { SIA_environment_previous_year_progress_net_change = current_year_net_gain }
}

SIA_change_specialist_amount = { 
	set_temp_variable = { nat_spec_check = change_temp }
	add_to_temp_variable = { nat_spec_check = SIA_specialists_var }
	
	if = {
		limit = { #Anti Cheat that kills a specialist in a state if national pool is zero
			check_variable = { change_temp < 0 }
			check_variable = { SIA_specialists_var = 0 }
		}
		custom_effect_tooltip = SIA_change_specialist_amount_tt_reduce
		SIA_for_loop_reduce_specialist_in_state = yes
	}
	else_if = {
		limit = {#there's a value in national var but it's less than the change
			check_variable = { nat_spec_check < 0 }
		}
		custom_effect_tooltip = SIA_change_specialist_amount_tt_reduce
		add_to_variable = { SIA_specialists_var = change_temp }
		set_temp_variable = { change_temp = SIA_specialists_var } #After adding, grab specialists since it'll be most likely negative and then do loop check
		
		SIA_for_loop_reduce_specialist_in_state = yes
	}
	else = { #normal adding
		add_to_variable = { SIA_specialists_var = change_temp }
		custom_effect_tooltip = SIA_change_specialist_amount_tt
	}
	clamp_variable = { #Specialists should never go to negative level
		var = SIA_specialists_var
		min = 0
	}
	SIA_calculate_total_montly_fuel_growth = yes
}
SIA_for_loop_reduce_specialist_in_state = {
	for_each_loop = { 
		array = SIA_map_var 
		value = v 
		
		var:v = {
			if = {
				limit = { #if there's a specialist and the change var is not 0, find and subtract
					check_variable = { SIA_specialists_in_state > 0 }
					NOT = {
						check_variable = { change_temp = 0 }
					}
				}
				subtract_from_variable = { SIA_specialists_in_state = 1 }
				add_to_temp_variable = { SIA.change_temp = 1 }
				add_to_variable = { SIA.SIA_specialists_var_temp = SIA_specialists_in_state }
			}
		}
	}
	if = {
		limit = { #if somehow the change var is alive and there's still a specialist somewhere, continue, if not then go to next effect
			NOT = {
				check_variable = { change_temp = 0 }
			}
			check_variable = { SIA_specialists_var_temp > 0 }
		}
		set_temp_variable = { SIA_specialists_var_temp = 0 }
		SIA_for_loop_reduce_specialist_in_state = yes
	}
	else = {
		#stop, just continue ahead
	}
}
