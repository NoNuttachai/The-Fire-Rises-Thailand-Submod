SIA_elections_translate_votes_to_seats = {
	for_each_loop = {
		array = SIA_party_election_array
		var:v = {
		
			set_variable = { party_seats_elections = party_votes }
			multiply_variable = { party_seats_elections = 5 }
			round_variable = party_seats_elections
		}
	}
}
